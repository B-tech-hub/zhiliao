const stable = (value) => JSON.stringify(canonical(value));
function canonical(value) {
  if (Array.isArray(value)) return value.map(canonical);
  if (value && typeof value === "object") return Object.fromEntries(Object.entries(value).sort(([a],[b]) => a.localeCompare(b)).map(([key,item]) => [key,canonical(item)]));
  return value;
}

async function invoke(args, timeoutMs = 15000) {
  const started = performance.now();
  try {
    const result = await runDocker(args, { env, timeoutMs });
    report.commands.push({ args, elapsedMs: Math.round(performance.now() - started), status: "passed" });

  } catch (error) {
    report.commands.push({ args, elapsedMs: Math.round(performance.now() - started), status: "failed", error: redact(error.message) });
    throw error;
  }
}

const lines = (value) => value.split(/\r?\n/).filter(Boolean).sort();
async function preservedResources() {
  const result = {};
  for (const type of ["container", "network", "volume"]) {
    const ids = lines(await invoke(type === "container" ? ["ps", "-aq", "--no-trunc"] :
      [type, "ls", "-q", ...(type === "network" ? ["--no-trunc"] : [])]));
    const values = ids.length ? JSON.parse(await invoke([type, "inspect", ...ids])) : [];
    result[type] = {};
    for (const item of values) {
      let projection = item;
      if (type === "container") projection = {
        Id: item.Id, Name: item.Name, Image: item.Image, Config: item.Config, HostConfig: item.HostConfig,
        Mounts: [...item.Mounts].sort((a,b) => a.Destination.localeCompare(b.Destination)),
        Networks: item.NetworkSettings.Networks, Ports: item.NetworkSettings.Ports, RestartCount: item.RestartCount,
        State: Object.fromEntries(["Status","Running","Paused","Restarting","OOMKilled","Dead","Pid","StartedAt","FinishedAt"].map(key => [key,item.State[key]])),
      };
      result[type][item.Id ?? item.Name] = sha(stable(projection));
    }
  }
  result.images = lines(await invoke(["image", "ls", "-aq", "--no-trunc"]));
  return canonical(result);
}
async function assertPortFree() {
  await new Promise((resolve,reject) => {
    const server = net.createServer();
    server.once("error", reject);
    server.listen({ host: "127.0.0.1", port: 3210, exclusive: true }, () => server.close(error => error ? reject(error) : resolve()));
  });
}
async function health() {
  return await new Promise((resolve,reject) => {
    const started = performance.now();
    const request = http.get("http://127.0.0.1:3210/api/healthz", { agent: false }, response => {
      let body = "";
      response.setEncoding("utf8");
      response.on("data", chunk => {
        body += chunk;
        if (body.length > 8192) request.destroy(new Error("Health response exceeded budget"));
      });
      response.on("end", () => resolve({ status: response.statusCode, body, elapsedMs: Math.round(performance.now() - started) }));
      response.on("error", reject);
    });
    request.setTimeout(10000, () => request.destroy(new Error("Health request timeout")));
    request.on("error", reject);
  });
}
