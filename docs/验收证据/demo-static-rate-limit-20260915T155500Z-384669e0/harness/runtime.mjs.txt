async function invoke(args, timeoutMs = 20000) {
  const started = performance.now();
  return await new Promise((resolve, reject) => {
    const child = spawn("docker", args, { cwd: root, env, windowsHide: true, stdio: "pipe" });
    let stdout = "", stderr = "", failed;
    const timer = setTimeout(() => { failed = new Error("Docker 调用超时"); child.kill(); }, timeoutMs);
    const collect = target => chunk => {
      if (target === "stdout") stdout += chunk; else stderr += chunk;
      if (stdout.length + stderr.length > 8 * 1024 * 1024) { failed = new Error("Docker 输出超出预算"); child.kill(); }
    };
    child.stdout.setEncoding("utf8").on("data", collect("stdout"));
    child.stderr.setEncoding("utf8").on("data", collect("stderr"));
    child.on("error", error => { clearTimeout(timer); reject(error); });
    child.on("close", exitCode => {
      clearTimeout(timer);
      const entry = { args, elapsedMs: Math.round(performance.now() - started), exitCode };
      if (stderr.trim()) entry.stderr = redact(stderr.trim());
      report.commands.push(entry);
      if (failed || exitCode !== 0) reject(failed || new Error("Docker 失败：" + redact(stderr)));
      else resolve(stdout.trim());
    });
    child.stdin.end();
  });
}

async function preservation() {
  const result = {};
  for (const type of ["container", "network", "volume"]) {
    const ids = lines(await invoke(type === "container" ? ["ps", "-aq", "--no-trunc"] : [type, "ls", "-q", ...(type === "network" ? ["--no-trunc"] : [])]));
    const values = ids.length ? JSON.parse(await invoke([type, "inspect", ...ids])) : [];
    result[type] = {};
    for (const item of values) {
      const projected = type === "container" ? {
        Id: item.Id, Name: item.Name, Image: item.Image, Config: item.Config, HostConfig: item.HostConfig,
        Mounts: [...item.Mounts].sort((a, b) => a.Destination.localeCompare(b.Destination)),
        Networks: item.NetworkSettings.Networks, Ports: item.NetworkSettings.Ports, RestartCount: item.RestartCount,
        State: Object.fromEntries(["Status", "Running", "Paused", "Restarting", "OOMKilled", "Dead", "Pid", "StartedAt", "FinishedAt"].map(key => [key, item.State[key]])),
      } : item;
      result[type][item.Id || item.Name] = sha(stable(projected));
    }
  }
  result.images = lines(await invoke(["image", "ls", "-aq", "--no-trunc"]));
  return canonical(result);
}
async function portFree() {
  await new Promise((resolve, reject) => {
    const server = net.createServer();
    server.once("error", reject);
    server.listen({ host: "127.0.0.1", port, exclusive: true }, () => server.close(error => error ? reject(error) : resolve()));
  });
}
const agent = new http.Agent({ keepAlive: true, maxSockets: 6 });
function request(requestPath, { method = "GET", headers = {}, body = "" } = {}) {
  return new Promise((resolve, reject) => {
    const start = performance.now();
    const req = http.request({ host: "127.0.0.1", port, path: requestPath, method, agent, headers: { "User-Agent": "zhiliao-static-repair", ...headers } });
    const timer = setTimeout(() => req.destroy(new Error("HTTP 总时限已到")), 10000);
    req.once("error", error => { clearTimeout(timer); reject(error); });
    req.once("response", res => {
      const chunks = [];
      let bytes = 0;
      res.on("data", chunk => {
        chunks.push(chunk); bytes += chunk.length;
        if (bytes > 4 * 1024 * 1024) req.destroy(new Error("HTTP 正文超出预算"));
      });
      res.once("error", error => { clearTimeout(timer); reject(error); });
      res.once("end", () => {
        clearTimeout(timer);
        const content = Buffer.concat(chunks);
        resolve({ path: requestPath, status: res.statusCode, contentType: res.headers["content-type"], bytes, sha256: sha(content), elapsedMs: Math.round(performance.now() - start), body: content.toString("utf8") });
      });
    });
    req.end(body);
  });
}
const compact = ({ body, ...rest }) => rest;
function marked(requestPath, scenario, index) {
  return requestPath + (requestPath.includes("?") ? "&" : "?") + "probe=" + runId + "&scenario=" + scenario + "&i=" + index;
}
async function burst(scenario, paths, count, expected = null) {
  const results = new Array(count);
  let next = 0;
  const start = performance.now();
  await Promise.all(Array.from({ length: 6 }, async () => {
    while (next < count) {
      const index = next++;
      const source = paths[index % paths.length];
      const result = await request(marked(source, scenario, index));
      if (result.status === 200 && expected) {
        assert.equal(result.sha256, expected[source].sha256, "静态正文与基线不一致");
        assert.equal(result.contentType, expected[source].contentType);
      }
      results[index] = compact(result);
    }
  }));
  const summary = { scenario, count, concurrency: 6, elapsedMs: Math.round(performance.now() - start), statuses: {} };
  for (const item of results) summary.statuses[item.status] = (summary.statuses[item.status] || 0) + 1;
  write(path.join(evidence, scenario + ".json"), { ...summary, requests: results });
  report.checks[scenario] = summary;
  return summary;
}

function matchRejections(value, summary) {
  const found = value.split(/\r?\n/).filter(line => line.includes("limiting requests,") && line.includes('zone "demo_req"') && line.includes("scenario=" + summary.scenario + "&"));
  assert.equal(found.length, summary.statuses[503] || 0, "503 与请求限流日志数量不一致：" + summary.scenario);
  summary.matchedRequestRejections = found.length;
}
async function runtimeIdentity() {
  const resources = await owned.list();
  await owned.assertOwned(resources);
  for (const [type, count] of [["container", 3], ["volume", 3], ["network", 2]]) assert.equal(resources.filter(x => x.type === type).length, count);
  const containers = JSON.parse(await invoke(["container", "inspect", ...resources.filter(x => x.type === "container").map(x => x.id)]));
  const services = Object.fromEntries(containers.map(item => [item.Config.Labels["com.docker.compose.service"], item]));
  for (const item of containers) assert.equal(item.State.Health.Status, "healthy");
  for (const name of ["app", "mockllm"]) {
    const item = services[name];
    assert.equal(item.Image, expectedImage);
    assert.deepEqual(Object.keys(item.NetworkSettings.Networks), [project + "-net"]);
    assert.equal(Object.values(item.NetworkSettings.Ports || {}).some(value => value?.length), false);
  }
  assert.deepEqual(services.ingress.NetworkSettings.Ports["8080/tcp"], [{ HostIp: "127.0.0.1", HostPort: String(port) }]);
  return Object.fromEntries(Object.entries(services).map(([name, item]) => [name, { id: item.Id, image: item.Image, startedAt: item.State.StartedAt, health: item.State.Health.Status }]));
}
