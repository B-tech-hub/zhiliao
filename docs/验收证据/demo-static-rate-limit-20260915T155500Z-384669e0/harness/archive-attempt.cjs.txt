const fs = require("node:fs");
const path = require("node:path");
const task = __dirname;
const file = path.join(task, "runtime-private.json");
const prior = JSON.parse(fs.readFileSync(file, "utf8"));
if (prior.report.cleanup.status !== "removed" || prior.report.preservation.status !== "passed") throw new Error("前轮未完成清理或保护核对");
const evidence = path.join("D:/ClaudeProjects/ai_acknowladge/docs/验收证据", "demo-static-rate-limit-" + prior.runId);
const archive = path.join(evidence, "attempt-1");
fs.mkdirSync(archive);
for (const item of fs.readdirSync(evidence, { withFileTypes: true })) {
  if (item.isFile()) fs.copyFileSync(path.join(evidence, item.name), path.join(archive, item.name));
}
fs.writeFileSync(path.join(archive, "attempt-notes.json"), JSON.stringify({
  status: "未进入限流对照，不计为产品验收",
  observationsFromToolOutput: [
    "首轮在字体样本发现时失败：登录 HTML 缺少 CSS 或字体样本。三个服务已健康，Nginx 语法通过，随后清理并核对既有资源保持。",
    "临时重置命令受到 PowerShell 的 Node 参数引号处理影响，解析失败；随后运行被重入保护拒绝，未再次启动容器。",
    "runtime.json 保留累积命令和最终重入错误；首轮字体错误以工具返回记录补充，未伪造原始报告。",
  ],
  adjustment: "改从 CSS 解析真实字体；临时辅助代码写入文件执行，修复目录归属检查后的重跑入口。",
}, null, 2) + "\n");
fs.copyFileSync(file, path.join(task, "attempt-1-private.json"));
for (const key of ["upAttempted", "report", "preserved", "compose", "servicesBefore"]) delete prior[key];
prior.attempt = 2;
fs.writeFileSync(file, JSON.stringify(prior, null, 2) + "\n");
