import json
import hashlib
from pathlib import Path
import sys
import urllib.request
from datetime import datetime, timezone

sys.stdout.reconfigure(encoding='utf-8')
base = Path(__file__).parent / 'evidence'
root = Path('D:/ClaudeProjects/ai_acknowladge')
queries = [
 ('GHSA-rfgv-xxqx-mfg5.json', 'https://api.github.com/advisories/GHSA-rfgv-xxqx-mfg5'),
 ('GHSA-w293-vg96-wgc3.json', 'https://api.github.com/advisories/GHSA-w293-vg96-wgc3'),
 ('undici-7.29.1.json', 'https://registry.npmjs.org/undici/7.29.1'),
 ('node-22.23.2-undici.json', 'https://raw.githubusercontent.com/nodejs/node/v22.23.2/deps/undici/src/package.json'),
]
records=[]
for name,url in queries:
    started=datetime.now(timezone.utc).isoformat()
    request=urllib.request.Request(url, headers={'User-Agent':'zhiliao-dependency-investigation','Accept':'application/json'})
    with urllib.request.urlopen(request,timeout=20) as response:
        data=response.read()
    (base/name).write_bytes(data)
    parsed=json.loads(data)
    records.append({'file':name,'url':url,'queried_utc':started,'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()})
    if name.startswith('GHSA'):
        print(json.dumps({k:parsed.get(k) for k in ['ghsa_id','cve_id','summary','severity','published_at','updated_at','vulnerabilities','description','references']},ensure_ascii=False),flush=True)
    else:
        print(json.dumps({k:parsed.get(k) for k in ['name','version','engines','dependencies','dist']},ensure_ascii=False),flush=True)
(base/'source-queries.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
lock=json.loads((root/'package-lock.json').read_text(encoding='utf-8'))['packages']
inventory={k:v for k,v in lock.items() if k.endswith('/undici') or k.endswith('/jsdom') or v.get('dependencies',{}).get('undici') or v.get('optionalDependencies',{}).get('undici')}
(base/'dependency-inventory.json').write_text(json.dumps(inventory,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(inventory,ensure_ascii=False))
