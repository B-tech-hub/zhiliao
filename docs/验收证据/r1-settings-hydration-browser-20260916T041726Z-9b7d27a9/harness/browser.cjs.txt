async function createSession(name, timezoneId) {
  const profile = path.join(task, "browser-" + name);
  assert.equal(fs.existsSync(profile), false, "浏览器目录必须是新的");
  const context = await chromium.launchPersistentContext(profile, {
    executablePath: "C:/Program Files/Google/Chrome/Application/chrome.exe",
    headless: true, viewport: { width: 1440, height: 900 }, timezoneId, locale: "zh-CN",
    colorScheme: "light", serviceWorkers: "block", acceptDownloads: true,
    args: ["--no-proxy-server", "--disable-background-networking", "--disable-component-update",
      "--disable-sync", "--no-first-run", "--no-default-browser-check"],
  });
  const page = context.pages()[0] || await context.newPage();
  const current = { name, timezoneId, context, page, gate: null };
  session = current;
  page.setDefaultTimeout(20000);
  page.setDefaultNavigationTimeout(30000);
  await context.route("**/*", async route => {
    const url = new URL(route.request().url());
    if (["http:", "https:"].includes(url.protocol) && url.origin !== origin) {
      report.blocked.push({ at: new Date().toISOString(), path: url.pathname, origin: url.origin, action: report.activeAction });
      await route.abort("blockedbyclient");
      return;
    }
    if (current.gate && route.request().resourceType() === "script") await current.gate.promise;
    await route.continue();
  });
  const cdp = await context.newCDPSession(page);
  await cdp.send("Network.enable");
  await cdp.send("Network.setCacheDisabled", { cacheDisabled: true });
  page.on("response", response => {
    const request = response.request();
    report.network.push({ at: new Date().toISOString(), session: name, action: report.activeAction,
      path: new URL(response.url()).pathname, status: response.status(),
      resourceType: request.resourceType(), method: request.method() });
  });
  page.on("requestfailed", request => report.requestFailures.push({
    at: new Date().toISOString(), session: name, action: report.activeAction,
    path: new URL(request.url()).pathname, resourceType: request.resourceType(), error: request.failure()?.errorText,
  }));
  page.on("pageerror", error => report.pageErrors.push({
    at: new Date().toISOString(), session: name, action: report.activeAction,
    name: error.name, message: redact(error.message), stack: redact(error.stack || ""),
    cause: error.cause ? redact(error.cause.stack || String(error.cause)) : null,
  }));
  page.on("console", message => {
    if (["error", "warning"].includes(message.type())) report.console.push({
      at: new Date().toISOString(), session: name, action: report.activeAction,
      type: message.type(), text: redact(message.text()), location: message.location(),
    });
  });
  report.activeAction = name + "-login";
  assert.equal((await page.goto(origin + "/login", { waitUntil: "networkidle" })).status(), 200);
  await page.getByPlaceholder("请输入访问密码", { exact: true }).fill(credentials.password);
  const [login] = await Promise.all([
    page.waitForResponse(response => new URL(response.url()).pathname === "/api/auth/login" && response.request().method() === "POST"),
    page.getByRole("button", { name: "登录", exact: true }).click(),
  ]);
  assert.equal(login.status(), 200);
  await page.waitForURL(origin + "/");
  await page.waitForLoadState("networkidle");
  const actual = await page.evaluate(() => ({
    timezone: Intl.DateTimeFormat().resolvedOptions().timeZone,
    offsetMinutes: new Date().getTimezoneOffset(), nowUtc: new Date().toISOString(),
    userAgent: navigator.userAgent, language: navigator.language,
  }));
  assert.equal(actual.timezone, timezoneId);
  report.sessions.push({ name, requestedTimezone: timezoneId, actual, freshProfile: true,
    loginStatus: login.status(), browserVersion: context.browser().version() });
  progress("session-ready", { name, timezone: actual.timezone });
  return current;
}

async function expectedTime(page, timestamp) {
  return page.evaluate(value => {
    if (value === null) return { text: "从未备份", nowUtc: new Date().toISOString(), timezone: Intl.DateTimeFormat().resolvedOptions().timeZone };
    const timezone = Intl.DateTimeFormat().resolvedOptions().timeZone;
    const parts = date => Object.fromEntries(new Intl.DateTimeFormat("en-GB", {
      timeZone: timezone, year: "numeric", month: "2-digit", day: "2-digit",
      hour: "2-digit", minute: "2-digit", hourCycle: "h23",
    }).formatToParts(date).map(part => [part.type, part.value]));
    const backup = parts(new Date(value)), now = parts(new Date());
    let text;
    if (backup.year === now.year && backup.month === now.month && backup.day === now.day) text = backup.hour + ":" + backup.minute;
    else if (backup.year === now.year) text = Number(backup.month) + "月" + Number(backup.day) + "日";
    else text = backup.year + "/" + backup.month + "/" + backup.day;
    return { text, nowUtc: new Date().toISOString(), timezone };
  }, timestamp);
}

async function layoutAndScreenshot(name, stage) {
  const page = session.page;
  const section = dataSection(page);
  await section.scrollIntoViewIfNeeded();
  await page.evaluate(() => document.fonts.ready);
  const layout = await section.evaluate(element => {
    const span = [...element.querySelectorAll("span")].find(item => item.textContent.startsWith("最近备份："));
    const button = [...element.querySelectorAll("button")].find(item => item.textContent.includes("立即备份"));
    const rect = target => {
      const r = target.getBoundingClientRect();
      return { left: r.left, right: r.right, top: r.top, bottom: r.bottom, width: r.width, height: r.height };
    };
    return { viewport: { width: innerWidth, height: innerHeight }, bodyScrollWidth: document.documentElement.scrollWidth,
      sectionWidth: element.clientWidth, sectionScrollWidth: element.scrollWidth,
      theme: document.documentElement.classList.contains("dark") ? "dark" : "light",
      section: rect(element), time: rect(span), button: rect(button), text: span.textContent,
      backupEnabled: !button.disabled };
  });
  assert.ok(layout.sectionScrollWidth <= layout.sectionWidth + 1, "数据区块横向溢出");
  assert.ok(layout.bodyScrollWidth <= layout.viewport.width + 1, "页面横向溢出");
  for (const field of ["time", "button"]) {
    assert.ok(layout[field].left >= -1 && layout[field].right <= layout.viewport.width + 1, field + " 超出视口");
  }
  assert.equal(layout.backupEnabled, true);
  const file = name + ".png";
  await page.screenshot({ path: path.join(evidence, file), fullPage: false, animations: "disabled" });
  const entry = { name, stage, file, sha256: sha(fs.readFileSync(path.join(evidence, file))), ...layout };
  report.screenshots.push(entry);
  save();
  return entry;
}

async function serverSettings(name, options = {}) {
  const page = session.page;
  report.activeAction = name;
  let release;
  if (options.placeholderScreenshot) {
    const promise = new Promise(resolve => { release = resolve; });
    session.gate = { promise };
  }
  let response, html, initial;
  try {
    const waitUntil = options.placeholderScreenshot ? "commit" : "networkidle";
    response = options.reload ? await page.reload({ waitUntil }) : await page.goto(origin + "/settings", { waitUntil });
    assert.equal(response.status(), 200);
    html = await response.text();
    const span = html.match(/<span[^>]*>最近备份：[\s\S]*?<\/span>/);
    assert.ok(span, "服务器 HTML 缺少最近备份节点");
    const values = [...new Set([...html.matchAll(/lastBackupAt\\*":(null|-?\d+(?:\.\d+)?)/g)].map(match => match[1]))];
    assert.equal(values.length, 1, "无法唯一读取服务端备份时间戳");
    lastBackupAt = values[0] === "null" ? null : Number(values[0]);
    const ssrText = span[0].replace(/<[^>]*>/g, "");
    assert.equal(ssrText, "最近备份：" + (lastBackupAt === null ? "从未备份" : "—"));
    if (options.noBackup) assert.equal(lastBackupAt, null);
    else if (options.hasBackup) assert.notEqual(lastBackupAt, null);
    assert.equal(html.includes(credentials.mockKey), false);
    initial = { status: response.status(), htmlSha256: sha(html), htmlBytes: Buffer.byteLength(html),
      timeSpanHtml: span[0], text: ssrText, backedUpAt: lastBackupAt };
    if (options.placeholderScreenshot) {
      await expect(backupText(page)).toHaveText(ssrText);
      initial.browserBeforeScripts = await backupText(page).textContent();
      await layoutAndScreenshot(options.placeholderScreenshot, "scripts-deferred-first-frame");
    }
  } finally {
    session.gate = null;
    release?.();
  }
  await page.waitForLoadState("networkidle");
  const expected = await expectedTime(page, lastBackupAt);
  await expect(backupText(page)).toHaveText("最近备份：" + expected.text);
  await expect(dataSection(page).getByRole("button", { name: "立即备份", exact: true })).toBeEnabled();
  await verifyConfiguration(page);
  const entry = { name, session: session.name, mode: options.reload ? "reload" : "direct",
    server: initial, client: { ...expected, text: await backupText(page).textContent() },
    configurationReadback: true, secretMasked: true, pageErrorCount: report.pageErrors.length,
    scriptsDeferredForFirstFrame: Boolean(options.placeholderScreenshot) };
  report.cases.push(entry);
  progress("settings-passed", { name, serverText: initial.text, clientText: entry.client.text });
}
