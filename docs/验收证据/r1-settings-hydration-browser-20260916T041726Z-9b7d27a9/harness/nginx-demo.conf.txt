pid /tmp/nginx.pid;

events {
  worker_connections 256;
}

http {
  access_log /dev/stdout;
  error_log /dev/stderr warn;
  # 空 key 不计入请求限流；仅豁免 Next 构建静态资源。
  map $uri $demo_req_key {
    default $binary_remote_addr;
    ~^/_next/static/ "";
  }
  limit_conn_zone $binary_remote_addr zone=demo_conn:10m;
  limit_req_zone $demo_req_key zone=demo_req:10m rate=10r/s;

  upstream zhiliao_app {
    server app:3000;
  }

  server {
    listen 8080;
    server_name _;

    # 普通请求保持较小上限；仅导入接口需要完整的 200 MiB 入口上限。
    client_max_body_size 8m;
    client_body_timeout 30s;
    limit_conn demo_conn 32;
    limit_req zone=demo_req burst=40 nodelay;
    proxy_http_version 1.1;
    proxy_request_buffering off;
    proxy_buffering off;
    proxy_read_timeout 300s;
    proxy_send_timeout 300s;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    location = /api/import {
      client_max_body_size 200m;
      proxy_pass http://zhiliao_app;
    }

    location = /api/uploads {
      client_max_body_size 25m;
      proxy_pass http://zhiliao_app;
    }

    location / {
      proxy_pass http://zhiliao_app;
    }
  }
}
