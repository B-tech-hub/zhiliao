function countFiles(dir: string): number {
  if (!fs.existsSync(dir)) return 0;
  let count = 0;
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) count += countFiles(full);
    else if (entry.isFile()) count += 1;
  }
  return count;
}

          transcriptionFile: null,

        index: row.item.index,
        title: row.item.title,
        sha256: row.item.sha256,
        modelCalled: true,
        outcome: error ? "failed" : outcome,
        error,
        preview,
        transcriptionFile,
      });
      fs.writeFileSync(ledgerPath, JSON.stringify([...ledger.values()], null, 2), "utf8");
      console.log(`transcribe ${row.item.index} ${error ? "failed" : outcome}`);
    }

      imageRange: { minLongEdge: Math.min(...longEdges), maxLongEdge: Math.max(...longEdges) },
      transcription: {

        calls: rows.filter((row) => row.modelCalled).length,
        outcomes: rows.map(({ title, sha256, ...row }) => ({ ...row, title, sha256 })),
      },
    });

    expect(quality).toHaveLength(20);
    expect(new Set(quality.map((row) => row.index)).size).toBe(20);
    for (const row of quality) expect(["可用", "要修", "不可用"]).toContain(row.verdict);

    for (const secret of secrets) expect(fs.readFileSync(reportPath, "utf8").includes(secret)).toBe(false);
  });
